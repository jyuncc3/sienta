🛠️ 最新 Apps Script 程式碼 (支援多工作表自動建立防呆)

請在試算表中貼上此代碼，並部署為「新版本」。程式會自動為您建立 Car 與 Motorcycle 兩份獨立工作表：

function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var result = {
    car: [],
    motorcycle: []
  };

  //---------------------------------
  // 汽車分頁處理 (無工作表時自動新增)
  //---------------------------------
  var carSheet = ss.getSheetByName("Car");
  if (!carSheet) {
    carSheet = ss.insertSheet("Car");
    carSheet.appendRow(["DATE","N.KM","NT/L","L"]);
  }
  var carValues = carSheet.getDataRange().getDisplayValues(); 
  for (var i = 1; i < carValues.length; i++) {
    if(!carValues[i][0] && !carValues[i][1]) continue;
    result.car.push({
      id: generateUUID(),
      date: String(carValues[i][0]).trim(),
      mileage: Number(String(carValues[i][1]).replace(/[^0-9.-]/g, '')) || 0,
      price: Number(String(carValues[i][2]).replace(/[^0-9.-]/g, '')) || 0,
      liters: Number(String(carValues[i][3]).replace(/[^0-9.-]/g, '')) || 0
    });
  }

  //---------------------------------
  // 機車分頁處理 (無工作表時自動新增)
  //---------------------------------
  var motorSheet = ss.getSheetByName("Motorcycle");
  if (!motorSheet) {
    motorSheet = ss.insertSheet("Motorcycle");
    motorSheet.appendRow(["DATE","N.KM","NT/L","L"]);
  }
  var motorValues = motorSheet.getDataRange().getDisplayValues(); 
  for (var i = 1; i < motorValues.length; i++) {
    if(!motorValues[i][0] && !motorValues[i][1]) continue;
    result.motorcycle.push({
      id: generateUUID(),
      date: String(motorValues[i][0]).trim(),
      mileage: Number(String(motorValues[i][1]).replace(/[^0-9.-]/g, '')) || 0,
      price: Number(String(motorValues[i][2]).replace(/[^0-9.-]/g, '')) || 0,
      liters: Number(String(motorValues[i][3]).replace(/[^0-9.-]/g, '')) || 0
    });
  }

  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);

  //---------------------------------
  // 汽車資料覆蓋
  //---------------------------------
  var carSheet = ss.getSheetByName("Car");
  if (!carSheet) {
    carSheet = ss.insertSheet("Car");
  }
  carSheet.clearContents();
  
  var carData = [["DATE","N.KM","NT/L","L"]];
  if (payload && payload.car && Array.isArray(payload.car)) {
    payload.car.forEach(function(r) {
      carData.push([
        String(r.date || "").trim(), 
        Number(r.mileage || 0), 
        Number(r.price || 0), 
        Number(r.liters || 0)
      ]);
    });
  }
  carSheet.getRange(1, 1, carData.length, 4).setValues(carData);

  //---------------------------------
  // 機車資料覆蓋
  //---------------------------------
  var motorSheet = ss.getSheetByName("Motorcycle");
  if (!motorSheet) {
    motorSheet = ss.insertSheet("Motorcycle");
  }
  motorSheet.clearContents();
  
  var motorData = [["DATE","N.KM","NT/L","L"]];
  if (payload && payload.motorcycle && Array.isArray(payload.motorcycle)) {
    payload.motorcycle.forEach(function(r) {
      motorData.push([
        String(r.date || "").trim(), 
        Number(r.mileage || 0), 
        Number(r.price || 0), 
        Number(r.liters || 0)
      ]);
    });
  }
  motorSheet.getRange(1, 1, motorData.length, 4).setValues(motorData);

  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    carCount: payload && payload.car ? payload.car.length : 0,
    motorcycleCount: payload && payload.motorcycle ? payload.motorcycle.length : 0
  }))
  .setMimeType(ContentService.MimeType.JSON);
}

// 智慧 UUID 生成輔助函數
function generateUUID() {
  try {
    return Utilities.getUuid();
  } catch(e) {
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
      var r = Math.random() * 16 | 0, v = c == 'x' ? r : (r & 0x3 | 0x8);
      return v.toString(16);
    });
  }
}
